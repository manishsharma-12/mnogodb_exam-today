Task 1: Count orders by status
db.orders.aggregate([
  {
    $group: {
      _id: "$status",
      count: { $sum: 1 }
    }
  },
  {
    $sort: {
      count: -1
    }
  }
])
Task 2: Filter using $match and $project
db.orders.aggregate([
  {
    $match: {
      status: "delivered"
    }
  },
  {
    $project: {
      _id: 0,
      customer: 1,
      item: 1,
      qty: 1
    }
  }
])

Task 3: Revenue per item
db.orders.aggregate([
  {
    $match: {
      status: "delivered"
    }
  },
  {
    $group: {
      _id: "$item",
      totalQty: { $sum: "$qty" },
      revenue: {
        $sum: {
          $multiply: ["$qty", "$price"]
        }
      }
    }
  },
  {
    $sort: {
      revenue: -1
    }
  }
])
Task 4: Spend per customer

db.orders.aggregate([
  {
    $match: {
      status: "delivered"
    }
  },
  {
    $group: {
      _id: "$customer",
      orders: { $sum: 1 },
      totalSpend: {
        $sum: {
          $multiply: ["$qty", "$price"]
        }
      }
    }
  },
  {
    $sort: {
      totalSpend: -1
    }
  }
])

Task 5: Find the best customer
db.orders.aggregate([
  {
    $match: {
      status: "delivered"
    }
  },
  {
    $group: {
      _id: "$customer",
      totalSpend: {
        $sum: {
          $multiply: ["$qty", "$price"]
        }
      }
    }
  },
  {
    $sort: {
      totalSpend: -1
    }
  },
  {
    $limit: 1
  }
])

Task 1: Add 2 copies using $inc

javascript
db.books.updateOne(
  { title: "Operating Systems" },
  { $inc: { copies: 2 } }
)


Task 2: Reduce copies by 1

javascript
db.books.updateOne(
  { title: "Database Systems" },
  { $inc: { copies: -1 } }
)


Task 3: Upsert a new book

javascript
db.books.updateOne(
  { title: "Artificial Intelligence" },
  {
    $set: {
      author: "Russell",
      copies: 4,
      status: "available"
    }
  },
  { upsert: true }
)


Task 4: Update using $set and $inc together

javascript
db.books.updateOne(
  { title: "Computer Networks" },
  {
    $set: {
      status: "available"
    },
    $inc: {
      copies: 3
    }
  }
)


Task 5: Delete one book

javascript
db.books.deleteOne({
  title: "Old Almanac 1990"
})


Task 6: Delete damaged or lost books

javascript
db.books.deleteMany({
  status: {
    $in: ["damaged", "lost"]
  }
})


Task 7: Display final data

javascript
db.books.find(
  {},
  {
    _id: 0,
    title: 1,
    copies: 1
  }
).sort({
  title: 1
})
